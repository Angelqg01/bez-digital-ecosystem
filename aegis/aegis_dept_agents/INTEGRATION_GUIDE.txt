# CAMBIOS EN aegis/main.py (4 líneas a añadir):
# =====================================================

# --- AÑADIR ESTOS IMPORTS al inicio de main.py ---
from dept_agents import DeptAgentManager, dept_agents_router

# --- AÑADIR ESTE ROUTER en la sección de app.include_router ---
app.include_router(dept_agents_router)

# --- AÑADIR en el startup event de FastAPI ---
@app.on_event("startup")
async def startup_event():
    # ... tu código de startup existente ...
    app.state.dept_manager = DeptAgentManager()
    await app.state.dept_manager.start_all()

# --- AÑADIR en el shutdown event ---
@app.on_event("shutdown")
async def shutdown_event():
    # ... tu código de shutdown existente ...
    if hasattr(app.state, "dept_manager"):
        await app.state.dept_manager.stop_all()
